cmake_minimum_required(VERSION 3.19)
project(PaperLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(paper STATIC Sheet.cpp)
target_include_directories(paper PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(paper-tests ../../tests/paper/paper_tests.cpp)
target_link_libraries(paper-tests PRIVATE paper)
add_executable(paper-bench ../../tests/paper/paper_bench.cpp)
target_link_libraries(paper-bench PRIVATE paper)
add_executable(paper-cycles ../../tests/paper/paper_cycles.cpp)
target_link_libraries(paper-cycles PRIVATE paper)
enable_testing()
add_test(NAME paper-core COMMAND paper-tests)
